SET ECHO ON
SET FEEDBACK ON

SET PAGESIZE 200

COLUMN granted_role FORMAT A25

PROMPT ===== 00. CONTEXTO =====
SELECT BANNER_FULL FROM V$VERSION;
SELECT SYS_CONTEXT('USERENV','CON_NAME') AS contenedor,
       SYS_CONTEXT('USERENV','SESSION_USER') AS usuario,
       TO_CHAR(SYSDATE,'YYYY-MM-DD HH24:MI:SS') AS fecha
FROM DUAL;
SELECT PARAMETER, VALUE FROM V$OPTION ORDER BY PARAMETER;

PROMPT ===== OBS-01. Rol DBA y archivo de contrasenas =====
SELECT GRANTEE, GRANTED_ROLE, ADMIN_OPTION FROM DBA_ROLE_PRIVS WHERE GRANTED_ROLE = 'DBA';
SELECT * FROM V$PWFILE_USERS;

PROMPT ===== OBS-02. Privilegios ANY =====
SELECT GRANTEE, PRIVILEGE, ADMIN_OPTION FROM DBA_SYS_PRIVS
WHERE PRIVILEGE LIKE '%ANY%'
  AND GRANTEE IN (SELECT USERNAME FROM DBA_USERS WHERE ORACLE_MAINTAINED = 'N')
ORDER BY GRANTEE, PRIVILEGE;

PROMPT ===== OBS-03. Cuentas con contrasena por defecto =====
SELECT * FROM DBA_USERS_WITH_DEFPWD;

PROMPT ===== OBS-04. Estado y uso de cuentas =====
SELECT USERNAME, ACCOUNT_STATUS, PROFILE, LAST_LOGIN, ORACLE_MAINTAINED
FROM DBA_USERS ORDER BY ORACLE_MAINTAINED, USERNAME;

PROMPT ===== OBS-05. Politica de contrasenas =====
SELECT PROFILE, RESOURCE_NAME, LIMIT FROM DBA_PROFILES
WHERE RESOURCE_TYPE = 'PASSWORD' ORDER BY PROFILE, RESOURCE_NAME;

PROMPT ===== OBS-06. Privilegios a PUBLIC y paquetes de salida =====
SELECT GRANTEE, OWNER, TABLE_NAME, PRIVILEGE FROM DBA_TAB_PRIVS
WHERE GRANTEE = 'PUBLIC'
  AND TABLE_NAME IN ('UTL_FILE','UTL_HTTP','UTL_SMTP','UTL_TCP');
SELECT * FROM DBA_DIRECTORIES;

PROMPT ===== OBS-07. Roles y privilegios para la revision de accesos =====
SELECT GRANTEE, GRANTED_ROLE FROM DBA_ROLE_PRIVS
WHERE GRANTEE IN (SELECT USERNAME FROM DBA_USERS WHERE ORACLE_MAINTAINED = 'N')
ORDER BY GRANTEE;

PROMPT ===== OBS-08. Inventario de columnas con comentario =====
SELECT OWNER, TABLE_NAME, COLUMN_NAME, COMMENTS FROM DBA_COL_COMMENTS
WHERE COMMENTS IS NOT NULL
  AND OWNER IN (SELECT USERNAME FROM DBA_USERS WHERE ORACLE_MAINTAINED = 'N');

PROMPT ===== OBS-09. Politicas de acceso (VPD) =====
SELECT * FROM DBA_POLICIES;

PROMPT ===== OBS-10. Cifrado en reposo (TDE) =====
SELECT * FROM V$ENCRYPTION_WALLET;
SELECT * FROM V$ENCRYPTED_TABLESPACES;
SELECT * FROM DBA_ENCRYPTED_COLUMNS;

PROMPT ===== OBS-11. Backups cifrados =====
SELECT RECID, ENCRYPTED FROM V$BACKUP_PIECE;

PROMPT ===== OBS-12. Enmascaramiento (Data Redaction) =====
SELECT * FROM REDACTION_POLICIES;

PROMPT ===== OBS-13. Auditoria activa y protegida =====
SELECT PARAMETER, VALUE FROM V$OPTION WHERE PARAMETER = 'Unified Auditing';
SELECT * FROM AUDIT_UNIFIED_ENABLED_POLICIES;
SELECT GRANTEE, GRANTED_ROLE FROM DBA_ROLE_PRIVS
WHERE GRANTED_ROLE IN ('AUDIT_ADMIN','AUDIT_VIEWER');

PROMPT ===== OBS-14. Eliminacion segura (papelera) =====
SELECT OWNER, OBJECT_NAME, ORIGINAL_NAME, DROPTIME FROM DBA_RECYCLEBIN;

PROMPT ===== OBS-15. Red y parcheo =====
SELECT NETWORK_SERVICE_BANNER FROM V$SESSION_CONNECT_INFO
WHERE SID = SYS_CONTEXT('USERENV','SID');
SELECT PATCH_ID, ACTION, STATUS, ACTION_TIME, DESCRIPTION FROM DBA_REGISTRY_SQLPATCH
ORDER BY ACTION_TIME DESC;

SPOOL OFF
PROMPT ===== FIN =====

       SYS_CONTEXT('USERENV','SESSION_USER') AS usuario,
       TO_CHAR(SYSDATE,'YYYY-MM-DD HH24:MI:SS') AS fecha
FROM DUAL;

CREATE PROFILE PERFIL_CONF LIMIT
  FAILED_LOGIN_ATTEMPTS    5
  PASSWORD_LOCK_TIME       1/24
  PASSWORD_LIFE_TIME       90
  PASSWORD_GRACE_TIME      7
  PASSWORD_REUSE_MAX       10
  PASSWORD_REUSE_TIME      365
  INACTIVE_ACCOUNT_TIME    120
  PASSWORD_VERIFY_FUNCTION ORA12C_VERIFY_FUNCTION;

ALTER USER PDBADMIN PROFILE PERFIL_CONF;
ALTER USER SYSTEM   PROFILE PERFIL_CONF;

REVOKE EXECUTE ON SYS.UTL_FILE FROM PUBLIC;
REVOKE EXECUTE ON SYS.UTL_HTTP FROM PUBLIC;
REVOKE EXECUTE ON SYS.UTL_SMTP FROM PUBLIC;
REVOKE EXECUTE ON SYS.UTL_TCP  FROM PUBLIC;

ALTER USER SYSRAC ACCOUNT LOCK;

PROMPT ===== OBS-03. Cuentas con contrasena por defecto =====
SELECT * FROM DBA_USERS_WITH_DEFPWD;

PROMPT ===== OBS-04. Estado de cuentas abiertas =====
SELECT USERNAME, ACCOUNT_STATUS, PROFILE FROM DBA_USERS
WHERE ACCOUNT_STATUS = 'OPEN' ORDER BY USERNAME;

PROMPT ===== OBS-05. Perfil PERFIL_CONF =====
SELECT PROFILE, RESOURCE_NAME, LIMIT FROM DBA_PROFILES
WHERE PROFILE = 'PERFIL_CONF' AND RESOURCE_TYPE = 'PASSWORD'
ORDER BY RESOURCE_NAME;

PROMPT ===== OBS-06. PUBLIC sobre paquetes de salida =====
SELECT GRANTEE, OWNER, TABLE_NAME, PRIVILEGE FROM DBA_TAB_PRIVS

  AND TABLE_NAME IN ('UTL_FILE','UTL_HTTP','UTL_SMTP','UTL_TCP');

SCRIPT 3: mejora_fase2a_datos_prueba.sql
Ejecución: @mejora_fase2a_datos_prueba.sql <clave_RRHH> <clave_APP_CONSULTA>
==================================================

SET ECHO ON
SET VERIFY OFF
SET FEEDBACK ON
SET LINESIZE 220
SET PAGESIZE 200
COLUMN cedula FORMAT A12
COLUMN nombre FORMAT A22
COLUMN correo FORMAT A28
COLUMN comments FORMAT A60
COLUMN policy_name FORMAT A20
COLUMN column_name FORMAT A12
COLUMN function_type FORMAT A15
COLUMN function_parameters FORMAT A30

SET ECHO OFF
CREATE USER RRHH IDENTIFIED BY "&1"
  DEFAULT TABLESPACE USERS QUOTA 50M ON USERS PROFILE PERFIL_CONF;
GRANT CREATE SESSION, CREATE TABLE TO RRHH;
CREATE USER APP_CONSULTA IDENTIFIED BY "&2" PROFILE PERFIL_CONF;
GRANT CREATE SESSION TO APP_CONSULTA;
SET ECHO ON

SPOOL evidencia_fase2a.txt

PROMPT ===== CONTEXTO =====
SELECT SYS_CONTEXT('USERENV','CON_NAME') AS contenedor,
       SYS_CONTEXT('USERENV','SESSION_USER') AS usuario,
       TO_CHAR(SYSDATE,'YYYY-MM-DD HH24:MI:SS') AS fecha
FROM DUAL;

SELECT USERNAME, ACCOUNT_STATUS, PROFILE FROM DBA_USERS WHERE USERNAME IN ('RRHH','APP_CONSULTA');

CREATE TABLE RRHH.EMPLEADOS (
  ID           NUMBER(6)     PRIMARY KEY,
  NOMBRE       VARCHAR2(60)  NOT NULL,
  CEDULA       VARCHAR2(9)   NOT NULL,
  CORREO       VARCHAR2(80),
  SALARIO      NUMBER(10,2),
  DEPARTAMENTO VARCHAR2(30)
);

INSERT INTO RRHH.EMPLEADOS VALUES (3, 'Marta Chaves Leon',  '303330333', 'marta.chaves@ejemplo.test',  780000, 'Recursos Humanos');
INSERT INTO RRHH.EMPLEADOS VALUES (4, 'Pablo Jimenez Cruz', '404440444', 'pablo.jimenez@ejemplo.test', 1100000, 'Gerencia');
INSERT INTO RRHH.EMPLEADOS VALUES (5, 'Sofia Araya Vega',   '505550555', 'sofia.araya@ejemplo.test',   690000, 'TI');
COMMIT;

GRANT SELECT ON RRHH.EMPLEADOS TO APP_CONSULTA;

COMMENT ON COLUMN RRHH.EMPLEADOS.NOMBRE  IS 'CLASIFICACION: RESTRINGIDO - dato personal';
COMMENT ON COLUMN RRHH.EMPLEADOS.CEDULA  IS 'CLASIFICACION: CONFIDENCIAL - dato personal (Ley 8968)';
COMMENT ON COLUMN RRHH.EMPLEADOS.CORREO  IS 'CLASIFICACION: RESTRINGIDO - dato personal';
COMMENT ON COLUMN RRHH.EMPLEADOS.SALARIO IS 'CLASIFICACION: CONFIDENCIAL - dato laboral';

BEGIN
  DBMS_REDACT.ADD_POLICY(
    object_schema       => 'RRHH',
    object_name         => 'EMPLEADOS',
    policy_name         => 'PR_EMPLEADOS_CONF',
    policy_description  => 'Enmascara salario y cedula',
    column_name         => 'SALARIO',
    function_type       => DBMS_REDACT.FULL,
    expression          => 'SYS_CONTEXT(''USERENV'',''SESSION_USER'') <> ''RRHH''');
END;
/

BEGIN
  DBMS_REDACT.ALTER_POLICY(
    object_schema       => 'RRHH',
    object_name         => 'EMPLEADOS',
    policy_name         => 'PR_EMPLEADOS_CONF',
    action              => DBMS_REDACT.ADD_COLUMN,
    column_name         => 'CEDULA',
    function_type       => DBMS_REDACT.PARTIAL,
    function_parameters => 'VVVVVVVVV,VVVVVVVVV,*,1,5');
END;
